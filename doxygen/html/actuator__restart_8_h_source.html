82inline bool open_file(
83 std::ifstream& is,
84 const std::string& chkname,
85 const std::string& label,
86 const amrex::Real time)
87{
88 is.open(file_name(chkname, label));
89 if (!is.good()) {
90 return false;
91 }
92 std::string word;
93 std::string label_in;
94 int version = 0;
95 amrex::Real time_in = 0.0_rt;
96 if (!(is >> word >> version) || (word != "kynema_sgf_actuator_restart") ||
97 (version != 1)) {
98 return false;
99 }
100 if (!(is >> word >> label_in) || (word != "label") || (label_in != label)) {
101 return false;
102 }
103 if (!(is >> word >> time_in) || (word != "time")) {
104 return false;
105 }
106 const auto tol =
107 1.0e-10_rt * amrex::max<amrex::Real>(1.0_rt, std::abs(time));
108 return std::abs(time_in - time) <= tol;
109}
